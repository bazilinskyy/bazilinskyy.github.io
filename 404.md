<script>
  // /publications/<anything> moved to /p/<anything>. Pages have real redirect
  // stubs; this catches the assets (.pdf, .jpg, .txt) that cannot have one.
  (function () {
    var old = '/publications/';
    var path = window.location.pathname;
    if (path.indexOf(old) === 0 && path.length > old.length) {
      window.location.replace('/p/' + path.slice(old.length) +
                              window.location.search + window.location.hash);
    }
  }());
</script>
